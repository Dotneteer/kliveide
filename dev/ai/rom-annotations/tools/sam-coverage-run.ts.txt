/** The program's line count, walking PROG's lines to the $FF end */
function lines(s: SamTestSession): number {
  let page = s.peek(0x5a9f) & 0x1f;
  let off = s.sysVarWord(0x5aa0) - 0x8000;
  let n = 0;
  for (;;) {
    if (off >= 0x4000) {
      page++;
      off -= 0x4000;
    }
    if (s.page(page, off) === 0xff || n > 500) return n;
    off += 4 + (s.page(page, off + 2) | (s.page(page, off + 3) << 8));
    n++;
  }
}

it("records the ROM's access profile over BASIC", async () => {
  const total = new Uint8Array(0x8000);
  const info: string[] = [];
  {
    const s = await createSamSession();
    s.machine.setProfiling(true, false);
    s.bootToBasic();
    const f = s.machine.readProfileFlags(0, 0x8000)!;
    for (let i = 0; i < f.length; i++) total[i] |= f[i] & 0x03;
  }
  for (const [n, snippet] of SNIPPETS.entries()) {
    const s = await createSamSession();
    s.bootToBasic();
    s.machine.setProfiling(true, false);
    s.machine.resetProfile();
    let ok = true;
    for (const line of snippet.lines ?? []) {
      const before = lines(s);
      s.typeKeys(line + "\n", { settle: 20 });
      if (lines(s) === before) {
        info.push(`#${n} REJECTED: ${line}`);
        ok = false;
        break;
      }
    }
    if (ok && snippet.lines) s.typeKeys("RUN\n", { settle: 250 });
    if (ok) for (const cmd of snippet.direct ?? []) s.typeKeys(cmd + "\n", { settle: 150 });
    info.push(`#${n} ERRNR=${s.peek(0x5c3a)} PPC=${s.sysVarWord(0x5c45)}`);
    const f = s.machine.readProfileFlags(0, 0x8000)!;
    for (let i = 0; i < f.length; i++) total[i] |= f[i];
  }
  writeFileSync(OUT + ".txt", info.join("\n"));
  writeFileSync(OUT, Buffer.from(total));
}, 600_000);
