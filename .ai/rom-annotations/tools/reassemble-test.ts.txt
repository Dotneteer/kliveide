// A throwaway check that a ROM sidecar still describes every byte: it is exported as Klive source and
// reassembled, and must give the ROM back byte for byte (what the IDE's Export as source and
// rom-ann-export do). Copy to test/reverse-eng/zz-reassemble.test.ts, run, delete:
//   SIDECAR=~/Klive/RomAnnotations/sp48.rom.dis ROM=sp48.rom OUT=<scratchpad>/export.kz80.asm \
//     npx vitest run --config build/vitest.config.ts --project node test/reverse-eng/zz-reassemble.test.ts
// OUT keeps the exported source for reading.
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { expect, it } from "vitest";
import { Z80Assembler } from "@main/z80-compiler/z80-assembler";
import { AssemblerOptions } from "@main/compiler-common/assembler-in-out";
import { knownRomPageKind, romPageIdentity } from "@common/roms/romIdentity";
import { parseAnnotations } from "@renderer/appIde/annotations/programAnnotations";
import { buildRomSourceExport, compareRomAssembly } from "@renderer/appIde/annotations/romSourceExport";
import { withInheritedEntries } from "@renderer/appIde/annotations/romInheritance";

it("reassembles to the ROM byte for byte", async () => {
  const ROMS = join(__dirname, "../../src/public/roms");
  const own = parseAnnotations(readFileSync(process.env.SIDECAR!, "utf8")).annotations!;
  const bytes = new Uint8Array(readFileSync(join(ROMS, process.env.ROM!))).subarray(0, 0x4000);
  // --- A sidecar that `inherits` is exported with its inherited, byte-bound entries, as the IDE does
  const { annotations } = await withInheritedEntries(own, () => bytes, async (sidecar, page) => ({
    annotations: parseAnnotations(readFileSync(join(ROMS, sidecar), "utf8")).annotations!,
    bytes: new Uint8Array(readFileSync(join(ROMS, sidecar.replace(/\.dis$/, "")))).subarray(page * 0x4000, (page + 1) * 0x4000)
  }));
  const pages = [{ page: 0, bytes, kind: knownRomPageKind(romPageIdentity(bytes)), crc32: romPageIdentity(bytes).crc32 }];
  const exported = await buildRomSourceExport(annotations, pages, { header: ["check"] });
  if (process.env.OUT) writeFileSync(process.env.OUT, exported.text);
  const output = await new Z80Assembler().compile(exported.text, new AssemblerOptions());
  const assembled = {
    errors: output.errors.map((e) => ({ line: e.line, message: e.message, isWarning: e.isWarning })),
    segments: output.segments.map((s) => ({ startAddress: s.startAddress, emittedCode: [...s.emittedCode] }))
  };
  expect(compareRomAssembly(exported.pages, assembled)).toEqual({ kind: "identical", bytes: bytes.length });
}, 120_000);
